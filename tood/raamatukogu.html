otsiAutoriJargi(autor) leiab kõik valitud autori raamatud. Kasutan filter(), sest samal autoril võib olla mitu raamatut.
Arvestan pärast 2000. aastat ilmunud väljaandeid. Aasta 2000 ei lähe arvesse. Kevade väljaande aasta muudan näites 2020-ks.
JavaScripti kood
// Sama funktsioon kuvab teksti konsoolis ja HTML-lehel.
function kuva(tekst) {
  console.log(tekst);
  if (typeof document !== "undefined") {
    document.getElementById("tulemused").textContent += tekst + "\n";
  }
}

const raamat = {
  pealkiri: "Kevade",
  autor: "Oskar Luts",
  aasta: 1912,
  kirjeldus() {
    return this.pealkiri + " – " + this.autor + " (" + this.aasta + ")";
  },
  muudaAasta(uusAasta) {
    this.aasta = uusAasta;
  }
};

kuva("RAAMAT");
kuva("Enne muutmist: " + raamat.kirjeldus());
// Muudame näites väljaande aastat, mitte teose esmailmumise aastat.
raamat.muudaAasta(2020);
kuva("Pärast muutmist: " + raamat.kirjeldus());

const raamatukogu = {
  raamatud: [
    raamat,
    { pealkiri: "Rehepapp", autor: "Andrus Kivirähk", aasta: 2000 },
    { pealkiri: "Mees, kes teadis ussisõnu", autor: "Andrus Kivirähk", aasta: 2007 }
  ],
  kuvaRaamatud() {
    this.raamatud.forEach((r, indeks) => {
      kuva((indeks + 1) + ". " + r.pealkiri + " – " + r.autor + " (" + r.aasta + ")");
    });
  },
  lisaRaamat(uusRaamat) {
    this.raamatud.push(uusRaamat);
    kuva("Lisatud: " + uusRaamat.pealkiri);
  },
  raamatuteArv() {
    const arv = this.raamatud.length;
    kuva("Raamatuid kokku: " + arv);
    return arv;
  },
  parast2000() {
    const arv = this.raamatud.filter(r => r.aasta > 2000).length;
    kuva("Pärast 2000. aastat ilmunud raamatuid: " + arv);
    return arv;
  },
  // Minu meetod otsib kõik ühe autori raamatud.
  otsiAutoriJargi(autor) {
    const leitud = this.raamatud.filter(r => r.autor === autor);
    kuva("Autor: " + autor);
    if (leitud.length === 0) kuva("Raamatuid ei leitud.");
    leitud.forEach(r => kuva(r.pealkiri + " (" + r.aasta + ")"));
    return leitud;
  }
};

kuva("\nRAAMATUKOGU ENNE LISAMIST");
raamatukogu.kuvaRaamatud();
raamatukogu.raamatuteArv();
kuva("\nUUE RAAMATU LISAMINE");
raamatukogu.lisaRaamat({
  pealkiri: "Apteeker Melchior ja Oleviste mõistatus",
  autor: "Indrek Hargla",
  aasta: 2010
});
kuva("\nRAAMATUKOGU PÄRAST LISAMIST");
raamatukogu.kuvaRaamatud();
raamatukogu.raamatuteArv();
raamatukogu.parast2000();
kuva("\nMINU MEETOD: OTSING AUTORI JÄRGI");
raamatukogu.otsiAutoriJargi("Andrus Kivirähk");
